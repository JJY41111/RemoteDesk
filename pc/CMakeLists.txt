cmake_minimum_required(VERSION 3.21)
project(RemoteDeskWindows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory(../out/deps/libdatachannel libdatachannel EXCLUDE_FROM_ALL)
add_subdirectory(../out/deps/opus opus EXCLUDE_FROM_ALL)
add_executable(remote_desk_client WIN32 client.cpp ../src/h264_network_decoder.cpp)
target_compile_definitions(remote_desk_client PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(remote_desk_client PRIVATE /W4 /permissive- /EHsc /utf-8 /O2)
target_include_directories(remote_desk_client PRIVATE ../src ../out/deps/libdatachannel/deps/json/single_include)
target_link_libraries(remote_desk_client PRIVATE datachannel opus winhttp crypt32 shlwapi shell32 comctl32 comdlg32 xinput
  user32 gdi32 gdiplus winmm ole32 oleaut32 mf mfplat mfreadwrite mfuuid wmcodecdspuuid)
set_target_properties(remote_desk_client PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../out")
add_executable(remote_desk_present_test EXCLUDE_FROM_ALL test/frame-surface.cpp)
target_link_libraries(remote_desk_present_test PRIVATE user32 gdi32)
add_executable(remote_desk_queue_test EXCLUDE_FROM_ALL test/video-queue.cpp)
target_compile_options(remote_desk_queue_test PRIVATE /UNDEBUG)
